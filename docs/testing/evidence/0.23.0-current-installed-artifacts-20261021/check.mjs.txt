import assert from 'node:assert/strict'
import { createHash } from 'node:crypto'
import { existsSync, lstatSync, readFileSync, readdirSync, writeFileSync } from 'node:fs'
import { dirname, join, resolve } from 'node:path'
import { InstallerProcesses } from '/home/sandbox/project/bazilion/scripts/testing/installer-process.mjs'

const control = '/control'
const npm = resolve(dirname(process.execPath), '../lib/node_modules/npm/bin/npm-cli.js')
const npmConfig = [`--userconfig=${control}/npm-user.rc`, `--globalconfig=${control}/npm-global.rc`]
const sha = (bytes) => createHash('sha256').update(bytes).digest('hex')
const save = (name, value) => writeFileSync(`${control}/${name}`, JSON.stringify(value, null, 2) + '\n', { flag: 'wx', mode: 0o600 })
const proc = new InstallerProcesses()
const result = { runtime: process.version, execPath: process.execPath, npmCli: npm, passed: false, stages: [], closure: null }
let observerPromise
let prefix
let text = ''
let observed = false

async function captureClosure() {
  assert.ok(prefix?.startsWith('/control/tmp/bazilion-installer-') && prefix.endsWith('/npm-prefix'))
  const before = lstatSync(prefix)
  const modules = join(prefix, 'lib/node_modules')
  const cli = join(modules, 'bazilion')
  const manifest = readFileSync(join(cli, 'package.json'))
  assert.equal(JSON.parse(manifest).version, '0.23.0')
  const inventory = []
  let bytes = 0
  function walk(dir, depth = 0) {
    assert.ok(depth < 40, 'manifest traversal depth bound')
    for (const e of readdirSync(dir, { withFileTypes: true })) {
      const path = join(dir, e.name)
      if (e.isDirectory()) walk(path, depth + 1)
      else if (e.isSymbolicLink()) {
        // npm executable symlinks are not a second dependency tree.
        assert.ok(path.includes('/.bin/'), `unexpected dependency symlink: ${path}`)
      } else if (e.isFile() && (e.name === 'package.json' || e.name === '.package-lock.json' || e.name.endsWith('.node'))) {
        const b = readFileSync(path)
        bytes += b.length
        assert.ok(bytes <= 64 * 1024 * 1024 && inventory.length < 10000, 'closure capture bound')
        inventory.push({ path: path.slice(prefix.length + 1), bytes: b.length, sha256: sha(b) })
        if (e.name === '.package-lock.json') writeFileSync(join(control, 'actual-prefix-package-lock.json'), b, { flag: 'wx', mode: 0o600 })
      }
    }
  }
  walk(modules)
  const listing = await proc.run(process.execPath, [npm, ...npmConfig, 'ls', '--global', '--all', '--omit=dev', '--json', '--offline', '--ignore-scripts', '--prefix', prefix], { env: process.env }, 120000)
  save('actual-prefix-npm-ls-outcome.json', listing)
  assert.equal(listing.code, 0, listing.stderr)
  assert.equal(listing.signal, null)
  assert.equal(listing.forced, false)
  assert.equal(listing.overflow, false)
  const tree = JSON.parse(listing.stdout)
  assert.equal(tree.dependencies.bazilion.version, '0.23.0')
  function problems(node) {
    assert.ok(!node.problems?.length, JSON.stringify(node.problems))
    for (const dependency of Object.values(node.dependencies ?? {})) problems(dependency)
  }
  problems(tree)
  const after = lstatSync(prefix)
  assert.equal(before.ino, after.ino)
  assert.equal(before.dev, after.dev)
  assert.equal(sha(readFileSync(join(cli, 'package.json'))), sha(manifest))
  const fact = { prefix, capturePoint: 'after admitted rebuild, while unchanged installer trial is running; before its final prefix removal', prefixIdentity: { dev: before.dev, ino: before.ino }, capturedBytes: bytes, inventory, npmTree: tree, passed: true }
  save('actual-prefix-closure.json', fact)
  return { passed: true, prefix, entries: inventory.length, capturedBytes: bytes }
}

try {
  const npmVersion = await proc.run(process.execPath, [npm, ...npmConfig, '--version'], { env: process.env })
  save('npm-version.json', npmVersion)
  assert.equal(npmVersion.code, 0)
  assert.equal(npmVersion.stdout.trim(), '11.19.0')
  const install = await proc.run(process.execPath, [npm, ...npmConfig, 'install', '--offline', '--ignore-scripts', '--engine-strict', '--no-audit', '--no-fund', '--prefix=/control/libraries', '/artifacts/bazilion-api-types-0.23.0.tgz', '/artifacts/bazilion-client-0.23.0.tgz'], { env: process.env }, 120000)
  save('libraries-install.json', install)
  assert.equal(install.code, 0, install.stderr)
  result.stages.push('actual current library tarballs installed offline/scripts-off outside workspace')
  const smoke = await proc.run(process.execPath, ['/control/libraries/smoke.mjs'], { env: process.env, cwd: '/control/libraries' }, 30000)
  save('libraries-smoke.json', smoke)
  assert.equal(smoke.code, 0, smoke.stderr)
  result.stages.push('library imports, two authenticated synthetic loopback requests and ApiClientError404 passed')
  // Observe, without altering the installer or its child arguments/oracles.
  const spawn = proc.spawn.bind(proc)
  proc.spawn = (...args) => {
    const child = spawn(...args)
    if (args[1]?.includes('/home/sandbox/project/bazilion/scripts/installer-e2e.mjs')) {
      child.stdout.on('data', (chunk) => {
        text = (text + chunk.toString()).slice(-65536)
        prefix ??= text.match(/^==> guarded npm install -g --ignore-scripts --prefix (\S+) /m)?.[1]
        if (!observed && text.includes('==> start deterministic provider and daemon on a fresh home')) {
          observed = true
          observerPromise = captureClosure().then(value => ({ value }), error => ({ error: String(error.stack ?? error) }))
        }
      })
    }
    return child
  }
  const trial = await proc.run(process.execPath, ['/home/sandbox/project/bazilion/scripts/installer-e2e.mjs', '/artifacts/bazilion-0.23.0.tgz'], { env: process.env, cwd: '/home/sandbox/project/bazilion' }, 1500000)
  save('installer-command.json', trial)
  const observer = observerPromise ? await observerPromise : { error: 'post-rebuild capture point was not observed' }
  save('closure-observer.json', observer)
  assert.equal(trial.code, 0, trial.stderr)
  assert.equal(trial.signal, null)
  assert.equal(trial.forced, false)
  assert.equal(trial.overflow, false)
  assert.ok(!observer.error, observer.error)
  const report = JSON.parse(readFileSync('/control/installer-evidence.json'))
  assert.equal(report.passed, true)
  assert.equal(report.fixtureRemoval.status, 'removed')
  assert.equal(existsSync(report.fixtureRoot), false)
  assert.equal(report.artifact.sha256, '9f7ee6ffba16532daa78744b31cfda3085d663187340f216e3fa9b4d8b0dc7ed')
  result.stages.push('unchanged guarded installer passed and its actual fixture root was removed')
  result.closure = observer.value
  result.passed = true
} catch (error) {
  result.error = String(error.stack ?? error)
  process.exitCode = 1
} finally {
  if (observerPromise) await observerPromise
  try { await proc.closeAll() } catch (error) { result.passed = false; result.cleanupError = String(error); process.exitCode = 1 }
  result.processes = proc.facts()
  result.limit = 'owned command/pipe and private namespace closure, not arbitrary-descendant or abnormal-host cleanup proof; local working tarballs, not final committed/public artifacts'
  save('outcomes.json', result)
  console.log(JSON.stringify({ passed: result.passed, stages: result.stages, closure: result.closure, error: result.error, cleanupError: result.cleanupError }))
}
