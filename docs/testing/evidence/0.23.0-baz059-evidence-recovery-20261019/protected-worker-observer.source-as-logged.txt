// Test-only observation of the actual serialized input. Never retains the input or credentials.
export function observeWorker(child,expected,env,log){
 const write=child.stdin.write.bind(child.stdin)
 let observed=false
 child.stdin.write=(chunk,...rest)=>{
  if(!observed){
   const input=JSON.parse(String(chunk))
   if(input.kind!==expected||input.imageGenerationEnabled!==true)throw Error('Trial worker posture mismatch')
   const selectedCredentialEnv=['OPENAI_API_KEY','OPENROUTER_API_KEY','ANTHROPIC_API_KEY','OPENAI_CODEX_OAUTH'].some(k=>!!env?.[k])
   if(expected==='protected'&&(selectedCredentialEnv||env?.NODE_OPTIONS))throw Error('Protected worker environment mismatch')
   log({kind:'worker_admission',surface:input.kind,imageEnabled:true,selectedCredentialEnv,trialPreloadInChild:!!env?.NODE_OPTIONS})
   observed=true
  }
  return write(chunk,...rest)
 }
}
